% рисование крестов
% https://tex.stackexchange.com/questions/123760/draw-crosses-in-tikz
\tikzset{
    cross/.pic = {
    \draw[line width=2.pt, rotate = 45] (-#1,0) -- (#1,0);
    \draw[line width=2.pt, rotate = 45] (0,-#1) -- (0, #1);
    }
}

            \foreach \j in {0,...,15}{
                \pgfmathsetmacro\cx{\i*0.25+0.125}
                \pgfmathsetmacro\cy{\j*0.25+0.125}
                \pgfmathsetmacro\dx{\cx-2.5}
                \pgfmathsetmacro\dy{\cy-2.1}
                \pgfmathsetmacro\u{\dx*cos(20)+\dy*sin(20)}
                \pgfmathsetmacro\v{-\dx*sin(20)+\dy*cos(20)}
                \pgfmathsetmacro\inside{((\u/1.3)^2+(\v/0.85)^2) < 1 ? 1 : 0}
                \ifdim \inside pt > 0pt
                    \draw [fill=red!25, draw=none] (\i*0.25,\j*0.25) rectangle (\i*0.25+0.25,\j*0.25+0.25);
                \fi
            }
