% один блок: два ReLU-нейрона и линейный сумматор, вместе считают T
\newcommand{\Tblock}[1]{
    \begin{scope}[xshift=#1]
        % нейроны с иконкой ReLU
        \draw [line width=1.pt] (0,0.45) rectangle (1.3,1.25);
        \draw [line width=1.pt] (0.25,0.65) -- (0.65,0.65) -- (1.05,1.05);
        \draw [line width=1.pt] (0,-1.25) rectangle (1.3,-0.45);
        \draw [line width=1.pt] (0.25,-1.05) -- (0.65,-1.05) -- (1.05,-0.65);
        % линейный сумматор
        \draw [line width=1.pt] (1.3,0.85) -- (2.2,0.1);
        \draw [line width=1.pt] (1.3,-0.85) -- (2.2,-0.1);
        \draw [line width=1.pt,fill=white] (2.35,0) circle (0.25cm) node {$+$};
        % стрелки на вход нейронов
        \draw [line width=1.pt] (-0.8,0) -- (-0.45,0);
        \draw [->, line width=1.pt] (-0.45,0) -- (-0.05,0.75);
        \draw [->, line width=1.pt] (-0.45,0) -- (-0.05,-0.75);
        % подпись блока
        \draw (1,-1.9) node {$T$};
    \end{scope}
}

        \draw [line width=1.pt] (11,0) -- (11.4,0);

        \draw [->, line width=1.pt] (13,0) -- (13.9,0) node[right] {$\hat y = T^{(k)}(x)$};
