\documentclass[border=10pt]{standalone}
\input{../dlbook_style}
\begin{document}
\begin{tikzpicture}[x=8mm, y=8mm]
  \tikzset{
    imgcell/.style={cell, fill=fillgreen},
    padcell/.style={cell, fill=white},
    wincell/.style={cell, fill=fillred},
    kercell/.style={cell, fill=fillblue},
    window/.style={draw=red, line width=2.2pt, line join=round},
  }
  \def\winrow{0}
  \def\panel#1#2#3{%
    \begin{scope}[xshift=#2 cm]
      \def\wincol{#1}
      \foreach \row [count=\i from 0] in {{0,0,0,0,0,0,0},{0,3,3,2,1,0,0},{0,0,0,1,3,1,0},{0,3,1,2,2,3,0},{0,3,1,2,2,3,0},{0,3,1,2,2,3,0},{0,0,0,0,0,0,0}}
        \foreach \v [count=\j from 0] in \row {
          \pgfmathtruncatemacro\inwin{and(and(\i>=\winrow, \i<\winrow+3), and(\j>=\wincol, \j<\wincol+3))}
          \pgfmathtruncatemacro\inimg{and(and(\i>=1, \i<=5), and(\j>=1, \j<=5))}
          \ifnum\inwin=1 \def\cs{wincell}\else\ifnum\inimg=1 \def\cs{imgcell}\else\def\cs{padcell}\fi\fi
          \node[\cs] at (\j, -\i) {$\v$};
        }
      \draw[window] (\wincol-0.5, -\winrow+0.5) rectangle (\wincol+2.5, -\winrow-2.5);
      \node[font=\Large] at (7.5, -3) {$\ast$};
      \foreach \row [count=\i from 2] in {{0,1,2},{2,2,0},{0,1,2}}
        \foreach \v [count=\j from 9] in \row
          \node[kercell] at (\j, -\i) {$\v$};
      \draw[window] (8.5, -1.5) rectangle (11.5, -4.5);
      \node[font=\Large, anchor=west] at (11.8, -3) {$= #3$};
    \end{scope}}
  \panel{0}{0}{6}
  \panel{1}{14}{14}
\end{tikzpicture}
\end{document}
